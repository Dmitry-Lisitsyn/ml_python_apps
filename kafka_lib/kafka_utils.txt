from aiokafka import AIOKafkaProducer, AIOKafkaConsumer
import json

KAFKA_BROKER = "localhost:9092"  # Замените на адрес вашего брокера Kafka
KAFKA_TOPIC_REQUESTS = "llm_prompts"  # Тема для запросов LLM
KAFKA_TOPIC_RESPONSES = "llm_responces"  # Тема для ответов LLM

# Инициализация Kafka producer
async def init_kafka_producer():
    producer = AIOKafkaProducer(bootstrap_servers=KAFKA_BROKER)
    await producer.start()
    return producer

# Закрытие Kafka producer
async def close_kafka_producer(producer):
    await producer.stop()

# Функция для отправки сообщения в Kafka
async def send_message_to_kafka(producer, message: dict, topic: str):
    # Десериализуем сообщение в строку перед отправкой
    await producer.send_and_wait(topic, json.dumps(message).encode("utf-8"))

# Инициализация Kafka consumer
async def init_kafka_consumer():
    consumer = AIOKafkaConsumer(
        KAFKA_TOPIC_REQUESTS,
        bootstrap_servers=KAFKA_BROKER,
        group_id="llm_group"
    )
    await consumer.start()
    return consumer

# Закрытие Kafka consumer
async def close_kafka_consumer(consumer):
    await consumer.stop()
